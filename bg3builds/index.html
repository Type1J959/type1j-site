How does the BG3 Team Build Planner prevent Act 1–3 gear conflicts across my party? ▾

In Baldur's Gate 3, many best-in-slot items like the Helmet of Arcane Acuity, Titanstring Bow, Luminous Armour, and Ring of Protection only exist once per playthrough. This planner automatically assigns unique Act 1, Act 2, and Act 3 gear across your 4 party members by role priority and selects the strongest available runner-up item whenever another companion already holds a unique piece.

Are the builds in this BG3 planner viable for Honor Mode and Tactician? ▾

Yes. Every preset role build includes a complete level 1–12 multiclass progression, starting stat spread, feat timing, and Act 1–3 gear upgrades specifically tuned for Honor Mode and Tactician runs, covering top meta archetypes alongside full custom build creation.

Can I save and share my Baldur's Gate 3 party builds? ▾

Yes. Your party automatically saves locally in your browser and syncs across devices with a free Type 1 J login. You can also export a formatted Markdown build summary or JSON file with one click to share with co-op friends or community Discords.

// HARDENED SERVER-SIDE AUTH + AUTO-MIGRATION + SAME-DEVICE AUTO-LOGIN (Code.gs)
// - Automatically migrates existing accounts from BG3_Data or 5-column BG3_Users (no accounts need to be remade!)
// - Random per-user cryptographic salt + 5,000-round HMAC-SHA256
// - Remembers a device's login token server-side so Google Sites refreshes stay logged in automatically!

function bytesToHex(bytes) {
  return bytes.map(function(b) {
    return ('0' + (b & 0xFF).toString(16)).slice(-2);
  }).join('');
}

function hashLegacyClientSha256(password) {
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'bg3_salt_v1_' + String(password));
  return bytesToHex(digest);
}

function hashPasswordServer(password, salt) {
  var sig = Utilities.computeHmacSha256Signature(String(password), String(salt));
  for (var i = 0; i < 5000; i++) {
    sig = Utilities.computeHmacSha256Signature(sig, Utilities.newBlob(String(salt)).getBytes());
  }
  return bytesToHex(sig);
}

function getDeviceFingerPrintHash(e, body) {
  var sig = String(body && body.deviceSig ? body.deviceSig : '');
  if (!sig) return '';
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'bg3_dev_' + sig);
  return bytesToHex(digest);
}

function ensureSheets(ss) {
  var uSheet = ss.getSheetByName('BG3_Users');
  if (!uSheet) {
    uSheet = ss.insertSheet('BG3_Users');
    uSheet.appendRow(['UserKey', 'Username', 'Salt', 'PasswordHash', 'SessionToken', 'UserDataJSON', 'UpdatedAt', 'DeviceFingerprint']);
  } else {
    // Auto-upgrade older 5-column BG3_Users sheet in place without losing any accounts!
    var headers = uSheet.getRange(1, 1, 1, Math.max(8, uSheet.getLastColumn())).getValues()[0];
    if (String(headers[2]) === 'PasswordHash' && String(headers[3]) === 'UserDataJSON') {
      var oldRows = uSheet.getDataRange().getValues();
      uSheet.clear();
      uSheet.appendRow(['UserKey', 'Username', 'Salt', 'PasswordHash', 'SessionToken', 'UserDataJSON', 'UpdatedAt', 'DeviceFingerprint']);
      for (var r = 1; r < oldRows.length; r++) {
        var uk = String(oldRows[r][0] || '').toLowerCase();
        if (!uk) continue;
        uSheet.appendRow([uk, oldRows[r][1] || uk, 'LEGACY_V1', oldRows[r][2] || '', Utilities.getUuid(), oldRows[r][3] || '{}', oldRows[r][4] || new Date().toISOString(), '']);
      }
    } else if (!headers[7]) {
      uSheet.getRange(1, 8).setValue('DeviceFingerprint');
    }
  }

  var cSheet = ss.getSheetByName('BG3_CommunityBuilds');
  if (!cSheet) {
    cSheet = ss.insertSheet('BG3_CommunityBuilds');
    cSheet.appendRow(['BuildId', 'Author', 'BuildName', 'BuildJSON', 'CreatedAt']);
  }

  // Also auto-migrate any accounts or community builds sitting in the original BG3_Data A1 cell
  var legacySheet = ss.getSheetByName('BG3_Data');
  if (legacySheet && uSheet.getLastRow() <= 1) {
    try {
      var raw = legacySheet.getRange('A1').getValue();
      if (raw) {
        var parsed = JSON.parse(raw);
        if (parsed && parsed.users && typeof parsed.users === 'object') {
          Object.keys(parsed.users).forEach(function(k) {
            var u = parsed.users[k];
            var uKey = String(k).toLowerCase();
            var uData = {
              parties: u.parties || [],
              activePartyId: u.activePartyId || null,
              customBuilds: u.customBuilds || [],
              customItems: u.customItems || []
            };
            uSheet.appendRow([uKey, u.username || k, 'LEGACY_V1', u.passwordHash || '', Utilities.getUuid(), JSON.stringify(uData), u.updatedAt || new Date().toISOString(), '']);
          });
        }
        if (parsed && Array.isArray(parsed.communityBuilds) && cSheet.getLastRow() <= 1) {
          parsed.communityBuilds.forEach(function(b) {
            if (b && b.id) cSheet.appendRow([b.id, b.author || '', b.name || '', JSON.stringify(b), new Date().toISOString()]);
          });
        }
      }
    } catch (err) {}
  }

  return { uSheet: uSheet, cSheet: cSheet };
}

function jsonResp(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function getCommunityBuildsList(cSheet) {
  var list = [];
  var rows = cSheet.getDataRange().getValues();
  for (var j = 1; j < rows.length; j++) {
    if (!rows[j][3]) continue;
    try { list.push(JSON.parse(rows[j][3])); } catch (err) {}
  }
  return list;
}

function findUserRow(uSheet, username) {
  var keyLower = String(username || '').trim().toLowerCase();
  if (!keyLower) return null;
  var rows = uSheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0] || '').toLowerCase() === keyLower) {
      return { rowNum: i + 1, cols: rows[i] };
    }
  }
  return null;
}

function findUserByDevice(uSheet, devFp) {
  if (!devFp) return null;
  var rows = uSheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][7] || '') === devFp && String(rows[i][4] || '')) {
      return { rowNum: i + 1, cols: rows[i] };
    }
  }
  return null;
}

function getHomeScheduleConfig(ss) {
  var hSheet = ss.getSheetByName('T1J_HomeSchedule');
  if (!hSheet) return null;
  var raw = hSheet.getRange('A1').getValue();
  if (!raw) return null;
  try { return JSON.parse(raw); } catch (err) { return null; }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ensureSheets(ss);
  return jsonResp({
    status: 'ok',
    communityBuilds: getCommunityBuildsList(sheets.cSheet),
    homeConfig: getHomeScheduleConfig(ss)
  });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ensureSheets(ss);
    var body = JSON.parse(e.postData ? e.postData.contents : '{}');
    var action = body.action || '';
    var username = String(body.username || '').trim();
    var uKey = username.toLowerCase();
    var devFp = getDeviceFingerPrintHash(e, body);

    if (action === 'resumeDeviceSession') {
      var devMatch = findUserByDevice(sheets.uSheet, devFp);
      if (!devMatch) {
        return jsonResp({ status: 'no_session', communityBuilds: getCommunityBuildsList(sheets.cSheet) });
      }
      var rData = {};
      try { rData = JSON.parse(devMatch.cols[5] || '{}'); } catch (err) {}
      return jsonResp({
        status: 'ok',
        username: String(devMatch.cols[1] || devMatch.cols[0]),
        sessionToken: String(devMatch.cols[4]),
        userData: rData,
        communityBuilds: getCommunityBuildsList(sheets.cSheet)
      });
    }

    if (action === 'logoutDevice') {
      var fOut = findUserRow(sheets.uSheet, username);
      if (fOut && String(fOut.cols[4] || '') === String(body.sessionToken || '')) {
        sheets.uSheet.getRange(fOut.rowNum, 5).setValue('');
        sheets.uSheet.getRange(fOut.rowNum, 8).setValue('');
      }
      return jsonResp({ status: 'ok' });
    }

    if (action === 'signup') {
      if (username.length < 2 || String(body.password || '').length < 3) {
        return jsonResp({ status: 'error', message: 'Invalid username or password length.' });
      }
      if (findUserRow(sheets.uSheet, username)) {
        return jsonResp({ status: 'error', message: 'That username is already taken.' });
      }
      var salt = Utilities.getUuid();
      var pwHash = hashPasswordServer(body.password, salt);
      var token = Utilities.getUuid() + '-' + Utilities.getUuid();
      var userData = body.userData || { parties: [], activePartyId: null, customBuilds: [], customItems: [] };
      var now = new Date().toISOString();
      var saveDevFp = body.persistSession !== false ? devFp : '';
      sheets.uSheet.appendRow([uKey, username, salt, pwHash, token, JSON.stringify(userData), now, saveDevFp]);
      return jsonResp({
        status: 'ok',
        username: username,
        sessionToken: token,
        userData: userData,
        communityBuilds: getCommunityBuildsList(sheets.cSheet)
      });
    }

    if (action === 'login') {
      var found = findUserRow(sheets.uSheet, username);
      if (!found) {
        return jsonResp({ status: 'error', message: 'No account found with that username.' });
      }
      var storedSalt = String(found.cols[2] || '');
      var storedHash = String(found.cols[3] || '');
      var isValid = false;

      if (storedSalt === 'LEGACY_V1' || !storedSalt) {
        // Seamless auto-upgrade for existing v1 accounts!
        var legacyCheck = hashLegacyClientSha256(body.password || '');
        if (storedHash && (legacyCheck === storedHash || String(body.password) === storedHash)) {
          isValid = true;
          var newSalt = Utilities.getUuid();
          var upgradedHash = hashPasswordServer(body.password || '', newSalt);
          sheets.uSheet.getRange(found.rowNum, 3, 1, 2).setValues([[newSalt, upgradedHash]]);
        }
      } else {
        var checkHash = hashPasswordServer(body.password || '', storedSalt);
        if (storedHash && checkHash === storedHash) isValid = true;
      }

      if (!isValid) {
        return jsonResp({ status: 'error', message: 'Incorrect password or PIN.' });
      }

      var newToken = Utilities.getUuid() + '-' + Utilities.getUuid();
      var loginDevFp = body.persistSession !== false ? devFp : '';
      sheets.uSheet.getRange(found.rowNum, 5).setValue(newToken);
      sheets.uSheet.getRange(found.rowNum, 7, 1, 2).setValues([[new Date().toISOString(), loginDevFp]]);
      var uData = {};
      try { uData = JSON.parse(found.cols[5] || '{}'); } catch (err) {}
      return jsonResp({
        status: 'ok',
        username: String(found.cols[1] || username),
        sessionToken: newToken,
        userData: uData,
        communityBuilds: getCommunityBuildsList(sheets.cSheet)
      });
    }

    if (action === 'getUserData') {
      var fGet = findUserRow(sheets.uSheet, username);
      if (!fGet || !body.sessionToken || String(fGet.cols[4] || '') !== String(body.sessionToken)) {
        return jsonResp({ status: 'unauthorized', message: 'Invalid or expired session token.' });
      }
      var parsedData = {};
      try { parsedData = JSON.parse(fGet.cols[5] || '{}'); } catch (err) {}
      return jsonResp({
        status: 'ok',
        username: String(fGet.cols[1] || username),
        userData: parsedData,
        communityBuilds: getCommunityBuildsList(sheets.cSheet)
      });
    }

    if (action === 'saveUserData') {
      var fSave = findUserRow(sheets.uSheet, username);
      if (!fSave || !body.sessionToken || String(fSave.cols[4] || '') !== String(body.sessionToken)) {
        return jsonResp({ status: 'unauthorized', message: 'Unauthorized: invalid session token.' });
      }
      var safeData = {
        parties: Array.isArray(body.userData?.parties) ? body.userData.parties : [],
        activePartyId: body.userData?.activePartyId || null,
        customBuilds: Array.isArray(body.userData?.customBuilds) ? body.userData.customBuilds : [],
        customItems: Array.isArray(body.userData?.customItems) ? body.userData.customItems : []
      };
      sheets.uSheet.getRange(fSave.rowNum, 6, 1, 2).setValues([[JSON.stringify(safeData), new Date().toISOString()]]);

      if (body.newCommunityBuild && body.newCommunityBuild.id) {
        var b = body.newCommunityBuild;
        b.author = String(fSave.cols[1] || username);
        var cRows = sheets.cSheet.getDataRange().getValues();
        var existingRow = 0;
        for (var k = 1; k < cRows.length; k++) {
          if (String(cRows[k][0]) === String(b.id)) {
            if (String(cRows[k][1]).toLowerCase() !== uKey) {
              return jsonResp({ status: 'unauthorized', message: 'Cannot overwrite another user build.' });
            }
            existingRow = k + 1;
            break;
          }
        }
        var bVals = [b.id, b.author, b.name || '', JSON.stringify(b), new Date().toISOString()];
        if (existingRow > 0) {
          sheets.cSheet.getRange(existingRow, 1, 1, 5).setValues([bVals]);
        } else {
          sheets.cSheet.appendRow(bVals);
        }
      }

      return jsonResp({
        status: 'ok',
        communityBuilds: getCommunityBuildsList(sheets.cSheet)
      });
    }

    if (action === 'saveHomeSchedule') {
      var expectedAdmin = 'a5da1cfdf7cc92b3f864b3bf48027a2fdb809610cd20bc7c299910cf4571044c';
      if (String(body.adminHash || '') !== expectedAdmin) {
        return jsonResp({ status: 'unauthorized', message: 'Invalid admin password.' });
      }
      var hSheet = ss.getSheetByName('T1J_HomeSchedule') || ss.insertSheet('T1J_HomeSchedule');
      hSheet.getRange('A1').setValue(JSON.stringify(body.homeConfig || {}));
      hSheet.getRange('A2').setValue('UpdatedAt: ' + new Date().toISOString());
      return jsonResp({ status: 'ok', homeConfig: body.homeConfig });
    }

    return jsonResp({ status: 'error', message: 'Unknown action.' });
  } finally {
    lock.releaseLock();
  }
}
